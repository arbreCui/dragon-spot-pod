def identity(s):
    return [getattr(s, 'st_' + key) for key in KEYS]


def kind(s):
    if stat.S_ISREG(s.st_mode):
        return 'file'
    if stat.S_ISDIR(s.st_mode):
        return 'dir'
    if stat.S_ISLNK(s.st_mode):
        return 'symlink'
    return 'other'


def pin(path):
    path = Path(path)
    before = identity(path.lstat())
    assert stat.S_ISREG(before[2]) and path.resolve(strict=True) == path
    h = hashlib.sha256()
    with path.open('rb') as stream:
        while chunk := stream.read(1024 * 1024):
            h.update(chunk)
    assert identity(path.lstat()) == before
    return {'path': str(path), 'sha256': h.hexdigest(), 'size': before[3], 'identity': before}


def load(path):
    p = pin(path)
    value = json.loads(Path(path).read_text())
    assert identity(Path(path).lstat()) == p['identity']
    return value, p


def keep(rel):
    return not (rel == '.git' or rel.startswith('.git/') or rel == R13_REL or rel.startswith(R13_REL + '/'))


def iscache(rel):
    return Path(rel).suffix in ('.pyc', '.pyo') or any(x in ('__pycache__', '.ruff_cache', '.pytest_cache', '.mypy_cache') for x in Path(rel).parts)


def fingerprint_from_rows(rows, excluded):
    """Reconstruct cleanup's serialized DFS stream without importing its source."""
    children = defaultdict(list)
    for rel, row in rows.items():
        if keep(rel):
            parent = Path(rel).parent.as_posix()
            children[parent].append((Path(rel).name, rel, row))
    h = hashlib.sha256()
    counts = {'regular_files': 0, 'symlinks': 0, 'directories': 0}
    stack = ['.']
    while stack:
        parent = stack.pop()
        h.update(json.dumps(['directory', parent], separators=(',', ':')).encode() + b'\n')
        counts['directories'] += 1
        dirs = []
        for name, rel, row in sorted(children[parent]):
            if row['type'] == 'dir':
                dirs.append(rel)
                continue
            if str(ROOT / rel) in excluded:
                continue
            ident = [row[key] for key in KEYS]
            if row['type'] == 'file':
                value = ['file', rel, ident]
                counts['regular_files'] += 1
            elif row['type'] == 'symlink':
                value = ['link', rel, ident, row['target']]
                counts['symlinks'] += 1
            else:
                continue
            h.update(json.dumps(value, separators=(',', ':')).encode() + b'\n')
        stack.extend(reversed(dirs))
    return {'sha256': h.hexdigest(), **counts}


def fresh_namespace(ns):
    root = Path(ns['root'])
    assert root.is_dir() and root.resolve(strict=True) == Path(ns['canonical_root'])
    files, dirs, links = set(), set(), {}
    stack = [root]
    while stack:
        parent = stack.pop()
        with os.scandir(parent) as entries:
            for entry in entries:
                p = Path(entry.path)
                rel = p.relative_to(root).as_posix()
                s = entry.stat(follow_symlinks=False)
                t = kind(s)
                if t == 'dir':
                    dirs.add(rel)
                    stack.append(p)
                elif t == 'file':
                    files.add(rel)
                elif t == 'symlink':
                    target = p.resolve(strict=True)
                    links[rel] = {'kind': 'directory' if target.is_dir() else 'file', 'target': str(target), 'text': os.readlink(p)}
                else:
                    raise ValueError('Nonordinary namespace member: ' + str(p))
    return {'root': str(root), 'regular_files': len(files), 'directories': len(dirs), 'symlinks': len(links), 'metadata_exact': files == set(ns['files']) and dirs == set(ns['directories']) and links == ns['symlinks']}
