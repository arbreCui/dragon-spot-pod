def identity(path):
    s = path.lstat()
    return [s.st_dev, s.st_ino, s.st_mode, s.st_size, s.st_mtime_ns, s.st_ctime_ns, s.st_nlink]

def pin(path, expected=None):
    path = Path(path)
    key = str(path)
    if key in PINS:
        d = PINS[key]
        assert identity(path) == d['identity'], ('changed cached identity', key)
    else:
        before = identity(path)
        assert stat.S_ISREG(before[2]), ('not regular', key)
        h = hashlib.sha256()
        with path.open('rb') as f:
            s = os.fstat(f.fileno())
            assert [s.st_dev, s.st_ino, s.st_mode, s.st_size, s.st_mtime_ns, s.st_ctime_ns, s.st_nlink] == before
            for block in iter(lambda: f.read(1024 * 1024), b''):
                h.update(block)
        assert identity(path) == before, ('changed full-read identity', key)
        d = dict(path=key, sha256=h.hexdigest(), identity=before, bytes=before[3], stable_full_read=True)
        PINS[key] = d
    if expected is not None:
        assert d['sha256'] == expected, ('expected SHA mismatch', key, expected, d['sha256'])
    return d

def read_json(path, expected=None):
    d = pin(path, expected)
    b = Path(path).read_bytes()
    assert hashlib.sha256(b).hexdigest() == d['sha256'] and identity(Path(path)) == d['identity']
    return json.loads(b)

def seal(folder, name='result.sha256'):
    p = folder / name
    pin(p)
    result = {}
    for line in p.read_text().splitlines():
        digest, relative = line.split(maxsplit=1)
        relative = relative.lstrip('*')
        target = folder / relative
        assert not Path(relative).is_absolute() and '..' not in Path(relative).parts
        assert target.is_relative_to(folder) and str(target) not in result
        assert re.fullmatch('[0-9a-f]{64}', digest)
        result[str(target)] = digest
    return result

def save(name, data):
    p = OUT / name
    assert not p.exists(), ('existing evidence', str(p))
    p.write_text(json.dumps(data, ensure_ascii=False, indent=2) + '\n')

def project_links():
    rows = []
    for parent, dirs, files in os.walk(ROOT, followlinks=False):
        for name in dirs + files:
            p = Path(parent) / name
            if p.is_symlink():
                rows.append([str(p), os.readlink(p), str(p.resolve())])
    rows.sort()
    return rows

def intersects(path, files, dirs):
    return path in files or any(Path(path).is_relative_to(Path(d)) for d in dirs)
