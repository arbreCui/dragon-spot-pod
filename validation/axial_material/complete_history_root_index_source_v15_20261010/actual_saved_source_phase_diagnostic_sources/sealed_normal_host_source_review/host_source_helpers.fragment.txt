def require(condition, reason):
    if not condition:raise ValueError(reason)

def unique_object(pairs):
    value={}
    for key,item in pairs:
        require(key not in value,'duplicate diagnostic metadata key '+key)
        value[key]=item
    return value

def strict_metadata(path):
    return json.loads(path.read_text(),object_pairs_hook=unique_object,
            parse_constant=lambda value: (_ for _ in ()).throw(ValueError('nonfinite diagnostic metadata')))

def diagnostic_identity(value):
    return {key:getattr(value,key) for key in IDENTITY_KEYS}

def metadata_inputs(contract):
    # Deliberately do not call regular-file-only support.pin on these literals:
    # the original producer's literal/target/fd/alias contract permits symlinks.
    records={}
    for role,entry in contract.items():
        path=Path(entry['path']);literal=diagnostic_identity(path.lstat());target=diagnostic_identity(path.stat())
        alias=os.readlink(path) if stat.S_ISLNK(literal['st_mode']) else None
        require(stat.S_ISREG(target['st_mode']),'diagnostic input target is not a regular file '+role)
        require(literal==diagnostic_identity(path.lstat()) and target==diagnostic_identity(path.stat()),
                'diagnostic input changed during metadata observation '+role)
        require(alias==(os.readlink(path) if path.is_symlink() else None),'diagnostic literal alias changed '+role)
        records[role]={'literal_7id':literal,'target_7id':target,'literal_symlink_text':alias}
    return records

def admitted_input_contract(support, source_pins):
    paths=[Path(path) for path,entry in source_pins.items() if Path(path).name=='input_contract.json' and entry['sha256']==CONTRACT_SHA]
    require(len(paths)==1,'one fixed original input contract required')
    record=strict_metadata(paths[0]);contract=record['inputs']
    require(record['schema']=='AXMAT_SAVED_SOURCE_PHASE_SOURCE_ONLY_INPUT_CONTRACT_V1' and
            record['numeric_payload_parsed'] is False and type(contract) is dict and len(contract)==1140,
            'complete fixed original source-only contract required')
    require(all(record[key] is None for key in ('future_actual_evaluator_pin','future_actual_host_pin',
            'future_actual_output_pins','future_actual_terminal_pin')),'source-only contract must not impersonate actual receipts')
    tree=ast.parse(TARGET.read_text());assignment={}
    for node in tree.body:
        if isinstance(node,ast.Assign):
            for name in node.targets:
                if isinstance(name,ast.Name):assignment[name.id]=node.value
    require(ast.literal_eval(assignment['SOURCE_STATUS'])=='READY','Root admitted READY source required')
    require(ast.literal_eval(assignment['WALL_SECONDS'])==60.0 and
            ast.literal_eval(assignment['RSS_LIMIT_MIB'])==512.0 and
            ast.literal_eval(assignment['THREADS'])==ast.literal_eval(assignment['CHILDREN'])==1 and
            ast.literal_eval(assignment['RETRIES'])==0,'immutable admitted producer limits required')
    node=assignment['INPUTS'];require(isinstance(node,ast.Dict),'fixed literal INPUTS table required')
    actual={}
    for key,value in zip(node.keys,node.values):
        role=ast.literal_eval(key)
        require(type(role) is str and role not in actual and isinstance(value,ast.Tuple) and len(value.elts)==2,
                'strict literal producer input entry required')
        call,digest=value.elts
        require(isinstance(call,ast.Call) and isinstance(call.func,ast.Name) and call.func.id=='Path' and
                len(call.args)==1 and not call.keywords,'fixed literal producer Path required')
        path=ast.literal_eval(call.args[0]);sha=ast.literal_eval(digest)
        require(type(path) is str and Path(path).is_absolute() and valid_sha(sha),'fixed literal path/SHA required')
        actual[role]={'path':path,'expected_sha256':sha}
    require(set(actual)==set(contract) and len(actual)==1140,'all1140 original roles must remain exact')
    for role,entry in contract.items():
        require(actual[role]=={'path':entry['path'],'expected_sha256':entry['expected_sha256']},
                'fixed producer and archived contract differ '+role)
    for prefix in ('capture_source_','capture_return_','capture_field_'):
        require({role for role in actual if role.startswith(prefix)}=={prefix+f'{group:03d}' for group in range(1,371)},
                'all370 source/return/field input roles required')
    require(sum(entry['observed_metadata_7id']['st_size'] for entry in contract.values())==96042876,
            'original full saved-source input byte scope differs')
    return contract,support.pin(paths[0],CONTRACT_SHA)

def valid_sha(value):
    return type(value) is str and len(value)==64 and all(char in '0123456789abcdef' for char in value)

def validate_identity_record(record,path,sha,observation,reason):
    keys={'path','sha256','size_bytes','literal_7id_before','literal_7id_after','target_7id_before',
          'target_7id_after','fd_7id_before','fd_7id_after','literal_symlink_text'}
    require(type(record) is dict and set(record)==keys and record['path']==path and record['sha256']==sha,
            'complete fresh EOF receipt path/SHA/key set differs '+reason)
    require(type(record['size_bytes']) is int and record['size_bytes']>=0,'fresh EOF byte count differs '+reason)
    for key in ('literal_7id_before','literal_7id_after','target_7id_before','target_7id_after','fd_7id_before','fd_7id_after'):
        value=record[key]
        require(type(value) is dict and set(value)==set(IDENTITY_KEYS) and
                all(type(value[item]) is int for item in IDENTITY_KEYS),'full typed seven-field identity required '+reason)
    require(record['literal_7id_before']==record['literal_7id_after']==observation['literal_7id'] and
            record['target_7id_before']==record['target_7id_after']==record['fd_7id_before']==record['fd_7id_after']==observation['target_7id'] and
            record['literal_symlink_text']==observation['literal_symlink_text'] and
            record['size_bytes']==record['target_7id_before']['st_size'],
            'actual literal/target/fd/alias/EOF identity differs '+reason)


def validate_input_receipts(identity_receipt,contract,before,after,target_observation):
    require(set(identity_receipt)=={'before','after','evaluator_before','evaluator_after'},'whole input receipt scope differs')
    require(set(identity_receipt['before'])==set(identity_receipt['after'])==set(contract) and len(contract)==1140,
            'all1140 before/after fresh EOF records required')
    require(before==after,'one unchanged outer metadata observation required for every input literal')
    for role,entry in contract.items():
        left=identity_receipt['before'][role];right=identity_receipt['after'][role]
        validate_identity_record(left,entry['path'],entry['expected_sha256'],before[role],role+' before')
        validate_identity_record(right,entry['path'],entry['expected_sha256'],after[role],role+' after')
        require(left==right and left['size_bytes']==entry['observed_metadata_7id']['st_size'],
                'fixed complete input changed across whole saved-source child '+role)
    require(sum(identity_receipt['before'][role]['size_bytes'] for role in contract)==96042876,
            'all1140 complete original input bytes required')
    for name in ('evaluator_before','evaluator_after'):
        validate_identity_record(identity_receipt[name],str(TARGET),TARGET_SHA,target_observation,name)
    require(identity_receipt['evaluator_before']==identity_receipt['evaluator_after'],'actual evaluator changed across child')

def joint_archive_bytes(support):
    # SOURCE and Root receipts live outside ART. Preserve the original ART-only
    # complete-history helper; account all new diagnostic material separately.
    roots=(HOST,OUTPUT,SOURCE_ARCHIVE,ROOT_RECEIPTS)
    total=support.archive_bytes(roots)
    if not any(TARGET.is_relative_to(root) for root in roots):total+=TARGET.lstat().st_size
    return total
